resolved = {}; conflicts = {}; low_confidence = {}
for gid, pl in proposals.items():
    cg = {}
    for (ch, co, ag) in pl:
        if ch not in cg: cg[ch] = {"HIGH":[], "MEDIUM":[], "LOW":[]}
        cg[ch][co].append(ag)
    if len(cg) == 1:
        ch = list(cg.keys())[0]; info = cg[ch]
        ha = list(set(info["HIGH"])); ma = list(set(info["MEDIUM"])); la = list(set(info["LOW"]))
        aa = ha+ma+la
        if len(ha)>=2: resolved[gid]=(ch,"CONFIRMED",aa)
        elif len(ha)>=1: resolved[gid]=(ch,"HIGH",aa)
        elif len(ma)>=2: resolved[gid]=(ch,"HIGH",aa)
        elif len(ma)>=1: resolved[gid]=(ch,"MEDIUM",aa)
        else: low_confidence[gid]=(ch,aa)
    else:
        bc=None; bs=-1; cd=[]
        for ch, info in cg.items():
            ha=list(set(info["HIGH"])); ma=list(set(info["MEDIUM"])); la=list(set(info["LOW"]))
            sc=len(ha)*3+len(ma)+len(la)*0.1
            cd.append((ch,ha,ma,la,sc))
            if sc>bs: bs=sc; bc=ch; ba=ha+ma+la
        conflicts[gid]=cd
        top=[d for d in cd if d[0]==bc][0]
        if len(top[1])>=2: resolved[gid]=(bc,"CONFIRMED",ba)
        elif len(top[1])>=1: resolved[gid]=(bc,"HIGH",ba)
        elif len(top[2])>=2: resolved[gid]=(bc,"HIGH",ba)
        elif len(top[2])>=1: resolved[gid]=(bc,"MEDIUM",ba)
        else: low_confidence[gid]=(bc,ba)

def tbl(d,r):
    items=sorted(d.items(),key=lambda x:int(x[0]))
    r.append("| Glyph | Char | Glyph | Char | Glyph | Char | Glyph | Char |")
    r.append("|-------|------|-------|------|-------|------|-------|------|")
    for i in range(0,len(items),4):
        row=""
        for j in range(4):
            if i+j<len(items): row+=f"| {items[i+j][0]} | {items[i+j][1]} "
            else: row+="| | "
        r.append(row+"|")
