def find_resource_file(idx):
    prefix = '{:04d}_'.format(idx)
    for fn in os.listdir(RES_DIR):
        if fn.startswith(prefix) and fn.endswith('.bin'):
            return os.path.join(RES_DIR, fn)
    return None

def extract_messages(data):
    first_ffff = None
    for off in range(0, len(data) - 1, 2):
        val = struct.unpack('>H', data[off:off+2])[0]
        if val == 0xFFFF:
            first_ffff = off
            break
    if first_ffff is None:
        return []
    stream_data = data[first_ffff:]
    n = len(stream_data) // 2
    vals = struct.unpack('>{}H'.format(n), stream_data[:n*2])
    messages = []
    cur = []
    for v in vals:
        if v == 0xFFFF:
            if cur:
                messages.append(cur)
            cur = []
        elif v == 0xFFFE:
            cur.append(0xFFFE)
        elif v >= 0xFFC0:
            cur.append(v)
        else:
            cur.append(v)
    if cur:
        messages.append(cur)
    return messages

def decode_message(glyphs):
    parts = []
    for g in glyphs:
        if g == 0xFFFE:
            parts.append('\n')
        elif g >= 0xFFC0:
            parts.append('<ctrl:{:04X}>'.format(g))
        elif str(g) in gmap:
            parts.append(gmap[str(g)])
        else:
            parts.append('[{:04X}]'.format(g))
    return ''.join(parts)

def categorize_resource(res_id, filename, messages):
    all_text = ''
    for msg in messages:
        all_text += decode_message(msg) + ' '
    if not messages:
        return 'EMPTY/NO_MSG'
    has_jp = any(c for c in all_text if '\u3040' <= c <= '\u309F' or '\u30A0' <= c <= '\u30FF' or '\u4E00' <= c <= '\u9FFF')
    has_ascii = any(c for c in all_text if 'a' <= c <= 'z')
    if any(kw in all_text for kw in ['ダンジョン', '迷宮', '階', '層', '扉', '罠']):
        return 'DUNGEON'
    elif any(kw in all_text for kw in ['攻撃', '戦闘', '魔法', '呪文']):
        return 'BATTLE'
    elif any(kw in all_text for kw in ['店', '宿', '買', '売', '泊']):
        return 'SHOP/INN'
    elif any(kw in all_text for kw in ['冒険', 'クエスト', '依頼', '報酬']):
        return 'QUEST'
    elif any(kw in all_text for kw in ['装備', '武器', '防具', '鎧']):
        return 'EQUIPMENT'
    elif has_jp and len(messages) > 3:
        return 'DIALOGUE'
    elif has_jp:
        return 'UI/SYSTEM'
    elif has_ascii and not has_jp:
        return 'ASCII_ONLY'
    else:
        return 'UNKNOWN'

    fp = find_resource_file(res_id)
    if fp is None:
        lines_out.append('--- Resource {}: FILE NOT FOUND ---'.format(res_id))
        continue
    filename = os.path.basename(fp)
    with open(fp, 'rb') as f:
        data = f.read()
    messages = extract_messages(data)
    category = categorize_resource(res_id, filename, messages)
    if category not in categories:
        categories[category] = []
    categories[category].append(res_id)
    total_messages += len(messages)
    if len(messages) >= 1:
        rich_resources.append((res_id, len(messages), filename))
    lines_out.append(sep2)
    lines_out.append('Resource {} ({}) - {} bytes - {} messages'.format(res_id, filename, len(data), len(messages)))
    lines_out.append('Category: {}'.format(category))
    lines_out.append(sep2)
    if not messages:
        lines_out.append('  (no decodable messages)')
        lines_out.append('')
        continue
    res_unknowns = set()
    for msg in messages:
        for g in msg:
            if g != 0xFFFE and g < 0xFFC0 and str(g) not in gmap:
                res_unknowns.add(g)
                total_unknown_glyphs.add(g)
    if res_unknowns:
        lines_out.append('  Unknown glyph IDs: {}'.format(sorted(res_unknowns)))
    show_count = min(len(messages), 30)
    for mi in range(show_count):
        decoded = decode_message(messages[mi])
        lines_out.append('  MSG[{}]: {}'.format(mi, decoded))
    if len(messages) > show_count:
        lines_out.append('  ... ({} more messages)'.format(len(messages) - show_count))
    lines_out.append('')
