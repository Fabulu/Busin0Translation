def strip_codes(text):
    return re.sub(r'\[[0-9A-Fa-f]{4}\]', '', text)

def apply_vocab(text):
    for jp, en in sorted(vocab.items(), key=lambda x: -len(x[0])):
        text = text.replace(jp, en)
    return text

def fallback_translate(jp):
    cleaned = strip_codes(jp)
    translated = apply_vocab(cleaned)
    if len(translated) > 60:
        translated = translated[:57] + '...'
    return translated

result = []
hand_count = 0
fallback_count = 0

for e in data:
    key = (e['resource'], e['msg_index'])
    if key in translations:
        en = translations[key]
        hand_count += 1
    else:
        en = fallback_translate(e['japanese'])
        fallback_count += 1
    result.append({
        "resource": e["resource"],
        "msg_index": e["msg_index"],
        "japanese": e["japanese"],
        "english": en
    })
